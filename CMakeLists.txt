cmake_minimum_required(VERSION 3.16)
project(PairLab VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(pairlab src/main.cpp)
add_executable(pairlab_tests tests/tests.cpp)
add_executable(pairlab_live src/live.cpp)
add_executable(pairlab_live_tests tests/live_tests.cpp)
foreach(target pairlab pairlab_tests pairlab_live pairlab_live_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /bigobj)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME research_and_accounting COMMAND pairlab_tests)

add_test(NAME live_scoring COMMAND pairlab_live_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/live_fixture.json")
